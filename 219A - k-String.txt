#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define Fast_IO ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
//functions
#define yes puts("YES")
#define no puts("NO")

int main() {
    Fast_IO
    ll k; cin>>k;
    unordered_map<char,ll> map;
    string s; cin>>s;
    for(char ch : s){
        map[ch]++;
    }
    ll ans=0;
    for(auto& pair : map){
        if (pair.second % k != 0){
            ans=-1;
            break;
        }
    }
    string a="";
    if (ans!=-1){
        for(auto& pair : map){
            a += string(pair.second / k, pair.first);
        }
        string result;
        for(int i = 0; i < k; i++){
            result += a;
        }
        cout<<result<<endl;
    }
    else{
        cout<<ans<<endl;
    }
    return 0;
}
