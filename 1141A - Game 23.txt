#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define Fast_IO ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
//functions
#define yes puts("YES")
#define no puts("NO")

int main() {
    Fast_IO
    ll n,m; cin>>n>>m;
    if (m%n!=0){
        cout<<-1<<endl;
        return 0;
    }
    ll k=m/n;
    ll ans=0;
    while(k%2==0){
        k/=2;
        ans++;
    }
    while(k%3==0){
        k/=3;
        ans++;
    }
    if (k==1){
        cout<<ans<<endl;
    }
    else{
        cout<<-1<<endl;
    }
}
