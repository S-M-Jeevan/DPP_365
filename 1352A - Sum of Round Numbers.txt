int main() {
    Fast_IO
    vector<ll> pre;
    ll incr=1;
    while(incr<10000){
        for(ll i=1;i<=9;i++){
            pre.push_back(i*incr);
            
        }
        incr*=10;
    }
    pre.push_back(10000);
    reverse(pre.begin(),pre.end());
    ll t;
    cin>>t;
    while(t--){
        ll n;
        cin>>n;
        ll sum=0;
        vector<ll> ans;
        for(ll i=0;i<pre.size();i++){
            if (sum+pre[i]<=n){
                if (sum==n){
                    break;
                }
                sum+=pre[i];
                ans.push_back(pre[i]);
            }
        }
        cout<<ans.size()<<endl;
        for(ll i=0;i<ans.size();i++){
            cout<<ans[i]<<" ";
        }
        cout<<endl;
    }
}
