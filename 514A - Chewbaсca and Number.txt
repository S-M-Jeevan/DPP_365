int main() {
    Fast_IO
    ll n; cin>>n;
    string ans;
    while(n>0){
        ll rem=n%10;
        ll dig=min(rem,9-rem);
        ans+=to_string(dig);
        n/=10;
    }
    reverse(ans.begin(),ans.end());
    if (ans[0]=='0') ans[0]='9';
    cout<<ans<<endl;
}
